cmake_minimum_required(VERSION 3.20)
project(MinneriteLlamaBridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_TOOLS_INSTALL ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" upstream)
add_subdirectory("${LLAMA_SOURCE}/tools/quantize" quantize)
add_library(aios-llama-bridge SHARED bridge.cpp)
target_link_libraries(aios-llama-bridge PRIVATE llama llama-common ggml)
target_compile_definitions(aios-llama-bridge PRIVATE AIOS_BACKEND_DIR="${CMAKE_INSTALL_PREFIX}/bin")
install(TARGETS aios-llama-bridge LIBRARY DESTINATION lib)
install(FILES bridge.h DESTINATION include)
add_executable(aios-bridge-contract contract.cpp)
target_link_libraries(aios-bridge-contract PRIVATE aios-llama-bridge)
target_compile_options(aios-bridge-contract PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME bounded-abi COMMAND aios-bridge-contract)
add_test(NAME reject-backend-override COMMAND aios-bridge-contract --reject-backend-override)
