cmake_minimum_required(VERSION 3.25)
project(aios_consent VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(consent-ui STATIC scope_dialog.cpp)
target_link_libraries(consent-ui PUBLIC Qt6::Widgets)
target_compile_options(consent-ui PRIVATE -Wall -Wextra -Werror)
add_executable(aios-scope-dialog main.cpp)
target_link_libraries(aios-scope-dialog PRIVATE consent-ui)
target_compile_options(aios-scope-dialog PRIVATE -Wall -Wextra -Werror)
install(TARGETS aios-scope-dialog RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  enable_testing()
  add_executable(consent-ui-test tests.cpp)
  target_link_libraries(consent-ui-test PRIVATE consent-ui Qt6::Test)
  target_compile_options(consent-ui-test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME consent-ui COMMAND consent-ui-test)
  install(TARGETS consent-ui-test RUNTIME DESTINATION libexec)
endif()
